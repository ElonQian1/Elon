我想要的是超强的情感女声，可以切换真实不同女声的，我现在应该怎么办？

# 第二步：新增 IndexTTS2 Worker

你现在已有 `server/tts_worker/edge_tts_worker.py`。

不要改它，新增一个：


大概结构是这样：

```python
# 文件名：server/tts_worker/expressive_tts_worker.py

from __future__ import annotations

import os
import uuid
from pathlib import Path
from threading import Lock
from typing import Dict, Optional

from fastapi import FastAPI, HTTPException
from fastapi.responses import FileResponse
from pydantic import BaseModel, Field

# 注意：
# 这个 import 需要你把 IndexTTS2 项目安装到当前 Python 环境里。
# 不同版本的 IndexTTS2 接口可能略有变化，如果报参数错误，
# 以你安装版本的 indextts.infer_v2.IndexTTS2.infer 签名为准。
from indextts.infer_v2 import IndexTTS2


class ExpressiveTtsRequest(BaseModel):
    text: str = Field(..., min_length=1, max_length=1200)
    voiceId: str = Field(default="female_warm", min_length=1, max_length=64)
    emotionId: str = Field(default="neutral", min_length=1, max_length=64)
    intensity: float = Field(default=0.65, ge=0.0, le=1.0)


BASE_DIR = Path(__file__).resolve().parent

MODEL_DIR = Path(os.getenv("INDEXTTS2_MODEL_DIR", "checkpoints")).resolve()
CFG_PATH = Path(os.getenv("INDEXTTS2_CFG_PATH", "checkpoints/config.yaml")).resolve()

VOICE_DIR = Path(
    os.getenv("ELON_TTS_VOICE_DIR", "server/assets/tts/voices")
).resolve()

EMOTION_DIR = Path(
    os.getenv("ELON_TTS_EMOTION_DIR", "server/assets/tts/emotions/female")
).resolve()

OUTPUT_DIR = Path(
    os.getenv("ELON_TTS_OUTPUT_DIR", "server/runtime/tts_outputs")
).resolve()

OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

VOICE_MAP: Dict[str, str] = {
    "female_warm": "female_warm/speaker.wav",
    "female_bright": "female_bright/speaker.wav",
    "female_mature": "female_mature/speaker.wav",
    "female_cool": "female_cool/speaker.wav",
    "female_sweet": "female_sweet/speaker.wav",
}

EMOTION_MAP: Dict[str, str] = {
    "neutral": "neutral.wav",
    "happy": "happy.wav",
    "shy": "shy.wav",
    "comfort": "comfort.wav",
    "sad": "sad.wav",
    "crying": "crying.wav",
    "angry": "angry_soft.wav",
    "excited": "excited.wav",
}


def resolve_voice_path(voice_id: str) -> Path:
    relative_path = VOICE_MAP.get(voice_id)

    if relative_path is None:
        raise HTTPException(
            status_code=400,
            detail=f"不支持的 voiceId：{voice_id}",
        )

    path = VOICE_DIR / relative_path

    if not path.exists() or not path.is_file():
        raise HTTPException(
            status_code=500,
            detail=f"voiceId={voice_id} 对应的 speaker.wav 不存在：{path}",
        )

    return path


def resolve_emotion_path(emotion_id: str) -> Optional[Path]:
    relative_path = EMOTION_MAP.get(emotion_id)

    if relative_path is None:
        # 情绪不认识时，不要直接失败，先降级 neutral。
        relative_path = EMOTION_MAP["neutral"]

    path = EMOTION_DIR / relative_path

    if not path.exists() or not path.is_file():
        # 如果情绪参考音频缺失，允许不用 emotion prompt。
        return None

    return path


def clamp(value: float, min_value: float, max_value: float) -> float:
    return max(min_value, min(value, max_value))


print("正在加载 IndexTTS2 强情绪 TTS 模型...")

TTS_MODEL = IndexTTS2(
    cfg_path=str(CFG_PATH),
    model_dir=str(MODEL_DIR),
    use_fp16=os.getenv("INDEXTTS2_USE_FP16", "true").lower() == "true",
    use_cuda_kernel=os.getenv("INDEXTTS2_USE_CUDA_KERNEL", "false").lower() == "true",
    use_deepspeed=os.getenv("INDEXTTS2_USE_DEEPSPEED", "false").lower() == "true",
)

print("IndexTTS2 强情绪 TTS 模型加载完成。")

MODEL_LOCK = Lock()
app = FastAPI(title="Elon Expressive Female TTS Worker")


@app.get("/health")
def health() -> Dict[str, object]:
    return {
        "ok": True,
        "engine": "indextts2",
        "voices": list(VOICE_MAP.keys()),
        "emotions": list(EMOTION_MAP.keys()),
    }


@app.post("/synthesize")
def synthesize(request: ExpressiveTtsRequest) -> FileResponse:
    text = request.text.strip()

    if not text:
        raise HTTPException(status_code=400, detail="text 不能为空。")

    speaker_path = resolve_voice_path(request.voiceId)
    emotion_path = resolve_emotion_path(request.emotionId)

    safe_intensity = clamp(request.intensity, 0.0, 1.0)

    output_path = OUTPUT_DIR / f"{uuid.uuid4().hex}.wav"

    try:
        with MODEL_LOCK:
            if emotion_path is not None:
                TTS_MODEL.infer(
                    spk_audio_prompt=str(speaker_path),
                    text=text,
                    output_path=str(output_path),
                    emo_audio_prompt=str(emotion_path),
                    emo_alpha=safe_intensity,
                    use_random=False,
                    verbose=True,
                )
            else:
                TTS_MODEL.infer(
                    spk_audio_prompt=str(speaker_path),
                    text=text,
                    output_path=str(output_path),
                    use_random=False,
                    verbose=True,
                )

    except Exception as error:
        raise HTTPException(
            status_code=500,
            detail=f"IndexTTS2 合成失败：{error}",
        ) from error

    if not output_path.exists():
        raise HTTPException(
            status_code=500,
            detail="IndexTTS2 推理完成，但没有生成音频文件。",
        )

    return FileResponse(
        path=str(output_path),
        media_type="audio/wav",
        filename=output_path.name,
        headers={
            "x-elon-tts-worker": "indextts2",
            "x-elon-tts-engine": "expressive",
            "x-elon-tts-voice-id": request.voiceId,
            "x-elon-tts-emotion-id": request.emotionId,
        },
    )
```

启动方式：



# 第八步：文本也要“适合表演”

强情绪 TTS 不是直接把普通 LLM 回复丢进去就完事。

普通文本：

```text

```

强情绪文本：

```text
你回来了？
真的……你真的回来了。
我刚刚还以为，你不会再理我了。
```

你需要在 TTS 前加一个轻量改写层：

```text
LLM 原始回复
  ↓
根据 emotionId 改写成适合朗读的短句
  ↓
TTS
```

例如 `crying` 风格：

```text
短句
省略号
重复
停顿
不要太长
```

`shy` 风格：

```text
轻一点
语气词
不要太直白
```

`cool` 风格：

```text
短句
克制
少感叹号
```

这一步会让“情绪拉满”效果提升非常明显。

但这是第二阶段。

### 1. APK


APK 不应该知道 Windows PC 的 IP，也不应该直接请求 Windows。

也就是说，Linux 是你的 **TTS Router / Gateway**。

Windows PC 是私有推理节点，负责：

```text

```

它应该只对 Linux 可见，不对 APK 用户可见。

# 最推荐的链路

我建议你做成这样：

```text
APK
  ↓
POST https://你的域名/api/voice/tts
  ↓
Linux Rust Server
  ├─ 普通模式 / 兜底
  │    ↓
  │  Edge TTS Worker 5010
  │
  └─ 强情绪女声模式
       ↓
     http://127.0.0.1:15020/synthesize
       ↓ 反向 SSH 隧道
     Windows PC 127.0.0.1:5020
       ↓
     IndexTTS2 / CosyVoice Worker
```

Linux 上看到的是：

```text
http://127.0.0.1:15020
```

但真实服务在 Windows：

```text
Windows 127.0.0.1:5020
```

这就是最简单、最安全的做法。

---

# 为什么推荐“Windows 主动连 Linux”

因为 Windows PC 大概率在家里或办公室网络后面，可能有 NAT、防火墙、动态 IP。不要让 Linux 主动找 Windows 公网 IP。

你应该让 Windows 主动建立一个出站连接到 Linux。这样 Windows 不需要开放公网端口。

可选方案有三个：

| 方案                        | 适合情况                        | 我的建议           |
| ------------------------- | --------------------------- | -------------- |
| **SSH 反向隧道**              | 你只有 1 台 Windows + 1 台 Linux | **最先用这个，简单直接** |
| **Tailscale / WireGuard** | 你以后有多台 Worker               | 更适合长期维护        |
| **Cloudflare Tunnel**     | 不想开入站端口，想走 Cloudflare       | 可用，但要做好访问控制    |

Tailscale 官方文档说明，大多数情况下不需要手动打开防火墙端口，它会用 NAT traversal 让设备互联；WireGuard 官方定位是简单、快速、现代的 VPN；Cloudflare Tunnel 也是出站连接模式，不需要公网 IP 和入站端口。([Tailscale][1])

说明 Linux 已经能通过隧道访问 Windows TTS Worker。

# Linux 服务器配置应该改成这样

你的 Linux Rust server 不要直接写死 Edge Worker。

建议加配置：

```bash
TTS_EDGE_WORKER_URL=http://127.0.0.1:5010
TTS_EXPRESSIVE_WORKER_URL=http://127.0.0.1:15020
TTS_ENABLE_EXPRESSIVE=true
TTS_DEFAULT_ENGINE=edge
TTS_EXPRESSIVE_TIMEOUT_MS=45000
TTS_EDGE_TIMEOUT_MS=15000
```

然后 `/api/voice/tts` 内部逻辑改成：

```text
如果 engine = expressive：
    优先请求 Windows 强情绪 Worker
    失败 / 超时 / 不健康：
        降级 Edge TTS

如果 engine = edge：
    直接请求 Edge TTS

如果 engine = auto：
    crying / shy / comfort / angry / excited 走 expressive
    neutral / 普通朗读走 edge
```

这样你就不会破坏现在已经上线的 Edge TTS。

# 一定要加 Worker Token

即使用了隧道，我也建议 Linux 调 Windows Worker 时加一个内部 token。

Windows Worker 要求请求头：

```text
x-elon-tts-worker-token: 一串很长的随机密钥
```

Linux 调用时带上这个 header。

这样即使某天端口配错了，也不至于任何人都能调用你的 Windows TTS 服务。

示例：

```python
# 文件名：server/tts_worker/worker_auth.py

from __future__ import annotations

import os

from fastapi import Header, HTTPException


WORKER_TOKEN = os.getenv("ELON_TTS_WORKER_TOKEN", "").strip()


def verify_worker_token(
    x_elon_tts_worker_token: str | None = Header(default=None),
) -> None:
    """
    校验 Linux 服务器调用 Windows TTS Worker 时携带的内部密钥。

    注意：
    这个 token 不给 APK。
    只在 Linux Server 和 Windows Worker 之间使用。
    """
    if not WORKER_TOKEN:
        raise HTTPException(
            status_code=500,
            detail="Windows TTS Worker 没有配置 ELON_TTS_WORKER_TOKEN。",
        )

    if x_elon_tts_worker_token != WORKER_TOKEN:
        raise HTTPException(
            status_code=401,
            detail="TTS Worker token 错误。",
        )
```

然后 Worker 接口：

```python
# 文件名：server/tts_worker/expressive_tts_worker.py

from __future__ import annotations

from fastapi import Depends, FastAPI

from worker_auth import verify_worker_token


app = FastAPI(title="Elon Expressive TTS Worker")


@app.post("/synthesize", dependencies=[Depends(verify_worker_token)])
def synthesize():
    # 这里放你原来的强情绪 TTS 合成逻辑
    pass
```

Windows 环境变量：

```powershell
setx ELON_TTS_WORKER_TOKEN "换成一串很长的随机字符串"
```

Linux 环境变量：

```bash
export ELON_TTS_WORKER_TOKEN="同一串很长的随机字符串"
```

# Windows 自动启动建议

你的 Windows PC 是关键节点，不能靠手动开两个窗口。

```powershell
$ProjectDir = "D:\rust\active-projects\elon cli\server\tts_worker"
```

# 大量用户时必须加队列

Windows PC 如果只有一张 GPU，不要允许很多请求同时打进去。

强情绪 TTS 不是 Edge TTS，它可能一次合成要几秒甚至更久。你要在 Linux 做限制：

```text
同时最多 1 到 2 个 expressive 请求
排队最多 N 个
超时自动降级 Edge
重复文本走缓存
```

推荐策略：

```text
普通用户：
    Edge TTS

会员 / 沉浸模式：
    Windows 强情绪 TTS

Windows 队列过长：
    自动降级 Edge TTS，并在日志里记录
```

否则一旦多个用户同时说话，Windows PC 会被打爆。

# 缓存 key 必须包含 engine

你现在有 Edge 和强情绪两套引擎，缓存 key 必须包含：

```text

emotionId
```

```text
用户选择 female_sweet + expressive

```

缓存 key 示例：

```text
tts:v2:engine=expressive:model=indextts2:voice=female_sweet:emotion=crying:intensity=0.78:textHash=xxxx
```
