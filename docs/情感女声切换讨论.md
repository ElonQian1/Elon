你现在这一步**不要先改模型，也不要先动后端 TTS 生成逻辑**。你的后端 catalog 已经把 5 个声线暴露出来了，真正缺的是 APK 端的“用户选择声线 → 保存 → TTS 请求优先使用这个 voiceId”。

最小正确路线是：

```text
1. APK 增加 tts_voice_id 偏好
2. APK 增加“声音设置 / 声线选择”入口
3. VoiceServerTtsPlayer.kt 发 /api/voice/tts 时，voiceId 优先用用户选择
4. VoiceTtsEmotion.kt 不再决定声线，只决定 emotionId / intensity
5. 悬浮球和聊天区都走同一套声线偏好
6. 检查 TTS 音频缓存 key，必须包含 voiceId
```

核心规则要改成这个：

```text
用户选择声线：female_sweet
用户文本情绪：伤心

最终请求：
voiceId = female_sweet
emotionId = sad / crying / melancholic
intensity = 0.7

而不是：
voiceId = female_mature 或 female_warm
```

也就是说：**声线归用户选，情绪归情绪规则选。**

### 1. 新增一个声线偏好文件

如果 `VoiceSpeaker.kt` 里已经有自己的 `SharedPreferences` 名字，你要把下面的 `PREFS_NAME` 改成它当前正在用的那个，避免出现“朗读开关存在 A 文件、声线存在 B 文件”的混乱。

```kotlin
// 文件名：VoiceTtsPreferences.kt

package com.elon.app

import android.content.Context
import android.content.SharedPreferences

object VoiceTtsPreferences {
    private const val PREFS_NAME = "voice_speaker_preferences"

    const val KEY_TTS_SPEAK_ENABLED = "tts_speak_enabled"
    const val KEY_TTS_VOICE_ID = "tts_voice_id"

    const val DEFAULT_VOICE_ID = "female_warm"

    private fun prefs(context: Context): SharedPreferences {
        return context.applicationContext.getSharedPreferences(
            PREFS_NAME,
            Context.MODE_PRIVATE
        )
    }

    fun isSpeakEnabled(context: Context): Boolean {
        return prefs(context).getBoolean(KEY_TTS_SPEAK_ENABLED, true)
    }

    fun setSpeakEnabled(context: Context, enabled: Boolean) {
        prefs(context)
            .edit()
            .putBoolean(KEY_TTS_SPEAK_ENABLED, enabled)
            .apply()
    }

    fun getSelectedVoiceId(context: Context): String {
        val savedVoiceId = prefs(context)
            .getString(KEY_TTS_VOICE_ID, null)
            ?.trim()

        if (savedVoiceId.isNullOrBlank()) {
            return DEFAULT_VOICE_ID
        }

        if (!VoiceTtsVoiceCatalog.isKnownVoiceId(savedVoiceId)) {
            return DEFAULT_VOICE_ID
        }

        return savedVoiceId
    }

    fun setSelectedVoiceId(context: Context, voiceId: String) {
        val safeVoiceId = voiceId.trim()

        if (!VoiceTtsVoiceCatalog.isKnownVoiceId(safeVoiceId)) {
            throw IllegalArgumentException("未知 TTS 声线 voiceId：$safeVoiceId")
        }

        prefs(context)
            .edit()
            .putString(KEY_TTS_VOICE_ID, safeVoiceId)
            .apply()
    }

    fun resetSelectedVoiceId(context: Context) {
        prefs(context)
            .edit()
            .remove(KEY_TTS_VOICE_ID)
            .apply()
    }
}
```

```kotlin
// 文件名：VoiceTtsVoicePicker.kt

package com.elon.app

import android.app.AlertDialog
import android.content.Context
import android.widget.Toast

object VoiceTtsVoicePicker {
    fun show(
        context: Context,

            .takeIf { it >= 0 }

            "${voice.displayName}\n${voice.description}"

            .setTitle("选择 AI 女声")

                VoiceTtsPreferences.setSelectedVoiceId(
                    context = context,
                    voiceId = selectedVoice.id
                )

                Toast.makeText(
                    context,
                    "已切换为：${selectedVoice.displayName}",
                    Toast.LENGTH_SHORT
                ).show()

                onVoiceChanged?.invoke(selectedVoice)

                dialog.dismiss()
            }
            .setNegativeButton("取消", null)
            .show()
    }
}
```

## 4. 改 VoiceTtsEmotion.kt：不要再返回 voiceId

你现在的问题在这里：`VoiceTtsEmotion.kt` 根据文本情绪自动映射到 `female_warm`、`female_bright`、`female_mature`。这会覆盖用户选择。

应该改成：**只返回情绪，不返回声线。**

```kotlin
// 文件名：VoiceTtsEmotion.kt

package com.elon.app

data class VoiceTtsEmotionResult(
    val emotionId: String,
    val intensity: Float
)

object VoiceTtsEmotion {
    fun detect(text: String): VoiceTtsEmotionResult {
        val normalizedText = text.trim()

        if (normalizedText.isBlank()) {
            return VoiceTtsEmotionResult(
                emotionId = "neutral",
                intensity = 0.35f
            )
        }

        return when {
            containsAny(
                normalizedText,
                listOf("哈哈", "开心", "太好了", "成功了", "喜欢", "好棒", "耶", "太开心")
            ) -> {
                VoiceTtsEmotionResult(
                    emotionId = "happy",
                    intensity = 0.75f
                )
            }

            containsAny(
                normalizedText,
                listOf("难过", "伤心", "失落", "想哭", "对不起", "舍不得", "离开", "不见了")
            ) -> {
                VoiceTtsEmotionResult(
                    emotionId = "sad",
                    intensity = 0.72f
                )
            }

            containsAny(
                normalizedText,
                listOf("生气", "讨厌", "为什么", "别这样", "够了", "烦", "闭嘴")
            ) -> {
                VoiceTtsEmotionResult(
                    emotionId = "angry",
                    intensity = 0.68f
                )
            }

            containsAny(
                normalizedText,
                listOf("害羞", "不好意思", "脸红", "别看我", "讨厌啦")
            ) -> {
                VoiceTtsEmotionResult(
                    emotionId = "shy",
                    intensity = 0.66f
                )
            }

            containsAny(
                normalizedText,
                listOf("别担心", "没关系", "我陪你", "慢慢来", "抱抱", "辛苦了")
            ) -> {
                VoiceTtsEmotionResult(
                    emotionId = "comfort",
                    intensity = 0.62f
                )
            }

            containsAny(
                normalizedText,
                listOf("真的", "不会吧", "什么", "居然", "天啊", "太突然")
            ) -> {
                VoiceTtsEmotionResult(
                    emotionId = "surprised",
                    intensity = 0.7f
                )
            }

            else -> {
                VoiceTtsEmotionResult(
                    emotionId = "neutral",
                    intensity = 0.45f
                )
            }
        }
    }

    private fun containsAny(text: String, keywords: List<String>): Boolean {
        return keywords.any { keyword ->
            text.contains(keyword, ignoreCase = true)
        }
    }
}
```

注意：`emotionId` 要跟服务端当前支持的值对上。如果服务端现在支持的是 `calm / happy / sad / angry / surprised`，那就把这里的 `comfort / shy` 映射成服务端已有值。

## 5. 改 VoiceServerTtsPlayer.kt：voiceId 优先读用户偏好

你现在发送 `/api/voice/tts` 的地方在：

```text
VoiceServerTtsPlayer.kt:129
```

那一段应该从“情绪规则给 voiceId”改成“偏好给 voiceId，情绪规则给 emotionId/intensity”。

核心逻辑应该像这样：

```kotlin
// 文件名：VoiceServerTtsPlayer.kt

package com.elon.app

import android.content.Context
import org.json.JSONObject

object VoiceServerTtsRequestBuilder {
    fun buildRequestJson(
        context: Context,
        text: String
    ): JSONObject {
        val selectedVoiceId = VoiceTtsPreferences.getSelectedVoiceId(context)
        val emotion = VoiceTtsEmotion.detect(text)

        return JSONObject().apply {
            put("text", text)

            // 关键点：
            // voiceId 固定使用用户选择，不再由情绪规则覆盖。
            put("voiceId", selectedVoiceId)

            // 情绪只负责表达方式。
            put("emotionId", emotion.emotionId)
            put("intensity", emotion.intensity.toDouble())
        }
    }
}
```

然后你原来大概有类似这种代码：

```kotlin
val emotionStyle = VoiceTtsEmotion.detect(text)

json.put("voiceId", emotionStyle.voiceId)
json.put("emotionId", emotionStyle.emotionId)
json.put("intensity", emotionStyle.intensity)
```

要改成：

```kotlin
val selectedVoiceId = VoiceTtsPreferences.getSelectedVoiceId(context)
val emotionStyle = VoiceTtsEmotion.detect(text)

json.put("voiceId", selectedVoiceId)
json.put("emotionId", emotionStyle.emotionId)
```

## 6. VoiceSpeaker.kt 负责统一入口

你现在用户能切的只有：

```text
tts_speak_enabled
```

下一步应该让 `VoiceSpeaker.kt` 同时管理：

```text
tts_speak_enabled
tts_voice_id
```

悬浮球和聊天区都不要自己直接决定声线。都应该走：

```text
VoiceSpeaker.speak(...)
```

里面再调用：

```text

VoiceTtsEmotion.detect(text)
VoiceServerTtsPlayer.play(...)
```

结构应该是：

```kotlin
// 文件名：VoiceSpeaker.kt

package com.elon.app

import android.content.Context

object VoiceSpeaker {
    fun isSpeakEnabled(context: Context): Boolean {
        return VoiceTtsPreferences.isSpeakEnabled(context)
    }

    fun setSpeakEnabled(context: Context, enabled: Boolean) {
        VoiceTtsPreferences.setSpeakEnabled(context, enabled)
    }

    fun getSelectedVoiceId(context: Context): String {
        return VoiceTtsPreferences.getSelectedVoiceId(context)
    }

    fun setSelectedVoiceId(context: Context, voiceId: String) {
        VoiceTtsPreferences.setSelectedVoiceId(context, voiceId)
    }

    fun speak(
        context: Context,
        text: String
    ) {
        if (!isSpeakEnabled(context)) {
            return
        }

        if (text.isBlank()) {
            return
        }

        // 这里继续调用你现有的服务端 TTS 播放器。
        // 重点是 VoiceServerTtsPlayer 内部必须使用 tts_voice_id。
        VoiceServerTtsPlayer.play(
            context = context,
            text = text
        )
    }
}
```

如果你现在的 `VoiceServerTtsPlayer.play(...)` 没有 `context` 参数，那就加上。因为读取用户偏好需要 `Context`。

## 7. 一定要检查音频缓存 key

这个很容易踩坑。

如果你现在缓存 TTS 音频只用了：

```text
text
```

或者：

```text
text + emotionId
```

那用户切换声线后，可能还会播放旧声线的缓存音频。

缓存 key 必须至少包含：

```text
text
voiceId
emotionId
intensity
```

推荐：

```kotlin
// 文件名：VoiceTtsCacheKey.kt

package com.elon.app

import java.security.MessageDigest

object VoiceTtsCacheKey {
    fun build(
        text: String,
        voiceId: String,
        emotionId: String,
        intensity: Float
    ): String {
        val raw = buildString {
            append("voiceId=")
            append(voiceId)
            append("|emotionId=")
            append(emotionId)
            append("|intensity=")
            append(String.format("%.2f", intensity))
            append("|text=")
            append(text)
        }

        return sha256(raw)
    }

    private fun sha256(value: String): String {
        val digest = MessageDigest.getInstance("SHA-256")
        val bytes = digest.digest(value.toByteArray(Charsets.UTF_8))

        return bytes.joinToString(separator = "") { byte ->
            "%02x".format(byte)
        }
    }
}
```

否则会出现这种 bug：

```text
用户选择：甜美陪伴
播放：你好呀

用户切换：冷淡女王
再次播放：你好呀

结果：仍然听到甜美陪伴
```

这不是 TTS 没切成功，而是缓存 key 没把 `voiceId` 放进去。
